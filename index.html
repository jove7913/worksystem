/** * NIFCO 도장 작업일지 - Apps Script API (Step 3) * 1단계 기능 전체를 구글 시트와 연결 */ const SHEET_NAMES = { WORKS: 'Works', EMPLOYEES: 'Employees', PARTS: 'Parts', HISTORY: 'History' }; const WORKS_HEADERS = ['id','workerId','workerName','process','partNo','partName','carModel','color','size','lot','startedAt','endedAt','reworkQty','scrapQty','status','approvedBy','approvedAt']; /* ============================================================ 진입점 ============================================================ */ function doGet(e) { try { const action = (e && e.parameter && e.parameter.action) || 'ping'; let result; if (action === 'ping') { result = { ok: true, message: 'NIFCO API is alive', time: new Date().toISOString() }; } else if (action === 'employees') { result = { ok: true, data: getEmployees() }; } else if (action === 'parts') { result = { ok: true, data: getParts() }; } else if (action === 'works') { result = { ok: true, data: getWorks(e.parameter) }; } else if (action === 'history') { result = { ok: true, data: getWorkHistory(e.parameter.workId) }; } else if (action === 'bootstrap') { // 로그인 직후 필요한 모든 마스터 + 작업 데이터를 한 번에 result = { ok: true, employees: getEmployees(), parts: getParts(), works: getWorks(e.parameter) }; } else { result = { ok: false, error: 'Unknown action: ' + action }; } return jsonOut(result); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } } function doPost(e) { // 동시 쓰기 방지 (최대 30초 대기) const lock = LockService.getScriptLock(); try { lock.waitLock(30000); } catch (lockErr) { return jsonOut({ ok: false, error: 'Server busy, please retry' }); } try { const body = JSON.parse(e.postData.contents || '{}'); const action = body.action; let result; switch (action) { case 'ping': result = { ok: true, message: 'POST received', received: body }; break; case 'createWork': result = createWork(body.payload); break; case 'submitForApproval': result = submitForApproval(body.payload); break; case 'approveWorks': result = approveWorks(body.payload); break; case 'cancelApproval': result = cancelApproval(body.payload); break; case 'updateWork': result = updateWork(body.payload); break; case 'replaceParts': result = replaceParts(body.payload); break; // 품목 마스터 CRUD case 'addPart': result = addPart(body.payload); break; case 'updatePart': result = updatePart(body.payload); break; case 'deletePart': result = deletePart(body.payload); break; // 직원 마스터 CRUD case 'addEmployee': result = addEmployee(body.payload); break; case 'updateEmployee': result = updateEmployee(body.payload); break; case 'deleteEmployee': result = deleteEmployee(body.payload); break; case 'replaceEmployees': result = replaceEmployees(body.payload); break; default: result = { ok: false, error: 'Unknown action: ' + action }; } return jsonOut(result); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } /* ============================================================ 마스터 조회 ============================================================ */ function getEmployees() { const sheet = sh(SHEET_NAMES.EMPLOYEES); const rows = sheet.getDataRange().getValues(); if (rows.length < 2) return []; const headers = rows[0]; return rows.slice(1) .filter(r => r[0] !== '' && r[0] !== null) .map(r => { const obj = rowToObject(headers, r); // empId를 문자열로 통일 obj.empId = String(obj.empId); return obj; }) .filter(e => e.active !== false); } function getParts() { const sheet = sh(SHEET_NAMES.PARTS); const rows = sheet.getDataRange().getValues(); if (rows.length < 2) return []; const headers = rows[0]; return rows.slice(1) .filter(r => r[0] !== '' && r[0] !== null) .map(r => { const obj = rowToObject(headers, r); if (typeof obj.colors === 'string') { obj.colors = obj.colors.split(',').map(c => c.trim()).filter(c => c); } else if (!Array.isArray(obj.colors)) { obj.colors = []; } return obj; }) .filter(p => p.active !== false); } /* ============================================================ 작업(Works) 조회 ============================================================ */ function getWorks(params) { params = params || {}; const sheet = sh(SHEET_NAMES.WORKS); const rows = sheet.getDataRange().getValues(); if (rows.length < 2) return []; const headers = rows[0]; let works = rows.slice(1) .filter(r => r[0] !== '' && r[0] !== null) .map(r => { const obj = rowToObject(headers, r); obj.workerId = String(obj.workerId); // 날짜 객체를 ISO 문자열로 ['startedAt','endedAt','approvedAt'].forEach(k => { if (obj[k] instanceof Date) obj[k] = obj[k].toISOString(); }); obj.reworkQty = Number(obj.reworkQty) || 0; works = works.filter(w => { if (!w.startedAt) return false; return new Date(w.startedAt) >= cutoff; }); } // 작업자 필터 (작업자 본인 화면에서 사용) if (params.workerId) { works = works.filter(w => w.workerId === String(params.workerId)); } return works; } /* ============================================================ 작업 등록 / 결재 / 수정 ============================================================ */ function createWork(payload) { if (!payload) return { ok: false, error: 'payload required' }; const sheet = sh(SHEET_NAMES.WORKS); const row = WORKS_HEADERS.map(h => { if (h === 'id') return id; if (h === 'status') return payload.status || 'draft'; if (h === 'reworkQty') return Number(payload.reworkQty) || 0; if (h === 'scrapQty') return Number(payload.scrapQty) || 0; if (h === 'approvedBy' || h === 'approvedAt') return ''; return payload[h] !== undefined ? payload[h] : ''; }); sheet.appendRow(row); appendHistory({ workId: id, by: payload.workerName + '(' + payload.workerId + ')', action: '작업 등록', detail: '재작업 ' + (payload.reworkQty||0) + ' / 스크랩 ' + (payload.scrapQty||0) + ' / 최종 ' + ((payload.reworkQty||0)+(payload.scrapQty||0)) }); return { ok: true, id: id }; } function submitForApproval(payload) { if (!payload || !payload.workerId || !Array.isArray(payload.ids)) { return { ok: false, error: 'workerId and ids[] required' }; } const sheet = sh(SHEET_NAMES.WORKS); const range = sheet.getDataRange(); const values = range.getValues(); const headers = values[0]; const idCol = headers.indexOf('id'); const statusCol = headers.indexOf('status'); const workerCol = headers.indexOf('workerId'); let updated = 0; for (let i = 1; i < values.length; i++) { const id = values[i][idCol]; if (payload.ids.indexOf(id) >= 0 && values[i][statusCol] === 'pending') { sheet.getRange(i + 1, statusCol + 1).setValue('approved'); } } return { ok: true, updated: updated }; } function cancelApproval(payload) { if (!payload || !payload.id || !payload.by) { return { ok: false, error: 'id and by required' }; } const sheet = sh(SHEET_NAMES.WORKS); const approvedAtCol = headers.indexOf('approvedAt'); for (let i = 1; i < values.length; i++) { if (values[i][idCol] === payload.id) { sheet.getRange(i + 1, statusCol + 1).setValue('draft'); sheet.getRange(i + 1, approvedByCol + 1).setValue(''); sheet.getRange(i + 1, approvedAtCol + 1).setValue(''); appendHistory({ workId: payload.id, by: payload.by, action: '결재 취소', detail: '' }); return { ok: true }; } } return { ok: false, error: 'not found' }; } function updateWork(payload) { if (!payload || !payload.id || !payload.by) { return { ok: false, error: 'id and by required' }; } const sheet = sh(SHEET_NAMES.WORKS); const values = sheet.getDataRange().getValues(); const headers = values[0]; const idCol = headers.indexOf('id'); for (let i = 1; i < values.length; i++) { if (values[i][idCol] === payload.id) { const changes = []; const fields = ['color','lot','reworkQty','scrapQty']; fields.forEach(f => { if (payload.changes && payload.changes[f] !== undefined) { const col = headers.indexOf(f); if (col >= 0) { const oldVal = values[i][col]; let newVal = payload.changes[f]; if (f === 'reworkQty' || f === 'scrapQty') newVal = Number(newVal) || 0; if (String(oldVal) !== String(newVal)) { sheet.getRange(i + 1, col + 1).setValue(newVal); changes.push(f + ' ' + oldVal + '→' + newVal); } } } }); if (changes.length > 0) { appendHistory({ workId: payload.id, by: payload.by, action: '관리자 수정', detail: changes.join(', ') }); } return { ok: true, changes: changes }; } } return { ok: false, error: 'not found' }; } /* ============================================================ 품목 마스터 일괄 교체 ============================================================ */ function replaceParts(payload) { if (!payload || !Array.isArray(payload.parts)) { return { ok: false, error: 'parts[] required' }; } const sheet = sh(SHEET_NAMES.PARTS); const values = sheet.getDataRange().getValues(); // 중복 체크 for (let i = 1; i < values.length; i++) { if (String(values[i][0]).trim().toUpperCase() === String(payload.partNo).trim().toUpperCase()) { return { ok: false, error: 'DUPLICATE_PART' }; } } sheet.appendRow([ payload.partNo, payload.partName, payload.carModel || '', Array.isArray(payload.colors) ? payload.colors.join(',') : (payload.colors || ''), payload.size || '', true ]); return { ok: true }; } function updatePart(payload) { if (!payload || !payload.partNo) { return { ok: false, error: 'partNo required' }; } const sheet = sh(SHEET_NAMES.PARTS); const values = sheet.getDataRange().getValues(); } } return { ok: false, error: 'NOT_FOUND' }; } /* ============================================================ 직원 마스터 개별 CRUD ============================================================ */ function addEmployee(payload) { if (!payload || !payload.empId || !payload.name || !payload.role) { return { ok: false, error: 'empId, name, role required' };
